# UI Shell Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development
> (recommended) or superpowers:executing-plans to implement this plan task-by-task.
> Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the first UI increment of Agenda ADHD - a Next.js 15 prototype at
`prototypes/ui-shell/` with a Welcome screen, radial navigation menu, six placeholder section
routes, and a development-only palette switcher.

**Architecture:** A server component (`app/page.tsx`) reads `searchParams` to derive the
initial view and passes it to `AppShell`, a client component that owns the
`'WELCOME' | 'MENU_OPEN' | 'MENU_CLOSED'` state machine. Section routes are separate Next.js
pages that do not use AppShell. All visual tokens live in `app/globals.css`; no palette or
animation values are generated by TypeScript at runtime. The `prevViewRef` pattern
distinguishes in-app transitions from mount scenarios for focus management.

**Tech Stack:** Next.js 15, React 19, TypeScript strict, Tailwind CSS v4, Vitest 2,
@testing-library/react 16, @testing-library/user-event 14, @testing-library/jest-dom 6

> All shell commands in this plan are run from inside `prototypes/ui-shell/` unless a step
> explicitly says otherwise.

```ts
// lib/constants.ts
import { SECTIONS } from './sections'
import type { SectionId } from './sections'

export const APP_NAME = 'agenda'

/** Per-item stagger delay in ms. Applied as inline --stagger per RadialMenuItem. Provisional. */
export const ITEM_STAGGER_MS = 40

// Angles are measured from 12 o'clock (top), clockwise.
// toRad subtracts 90 degrees so 0 deg points up, not right.
const toRad = (deg: number) => (deg - 90) * (Math.PI / 180)

interface MenuItem {
  id:     SectionId
  label:  string
  angle:  number
  txUnit: number
  tyUnit: number
}

const MENU_ITEM_ANGLES: Array<{ id: SectionId; angle: number }> = [
  { id: 'oggi',       angle:   0 },   // 12h
  { id: 'calendario', angle:  60 },   // 2h
  { id: 'lavagna',    angle: 120 },   // 4h
  { id: 'inbox',      angle: 180 },   // 6h
  { id: 'quaderno',   angle: 240 },   // 8h
  { id: 'esercizi',   angle: 300 },   // 10h
]

// Unit-circle offsets (radius = 1.0). CSS --radial-radius multiplies these.
// Computed at module-load time; no runtime Math.cos/sin in JSX.
export const MENU_ITEMS: readonly MenuItem[] = MENU_ITEM_ANGLES.map(({ id, angle }) => {
  const section = SECTIONS.find(s => s.id === id)!
  return {
    id,
    label:  section.label,
    angle,
    txUnit: parseFloat(Math.cos(toRad(angle)).toFixed(6)),
    tyUnit: parseFloat(Math.sin(toRad(angle)).toFixed(6)),
  }
})
```

```ts
// __tests__/navigation-parity.test.ts
// @vitest-environment node

import { describe, it, expect } from 'vitest'
import { existsSync } from 'fs'
import { resolve } from 'path'
import { SECTIONS } from '@/lib/sections'
import { MENU_ITEMS } from '@/lib/constants'

describe('Section / menu parity', () => {
  it('SECTIONS and MENU_ITEMS have identical id sets', () => {
    const sectionIds = [...SECTIONS].map(s => s.id).sort()
    const menuIds    = [...MENU_ITEMS].map(m => m.id).sort()
    expect(menuIds).toEqual(sectionIds)
  })

  it('MENU_ITEMS contains exactly the same ids as SECTIONS in the same order', () => {
    // This test checks order - do NOT sort before comparing.
    // If both arrays are sorted inside the assertion, the test becomes a no-op
    // for order and only verifies set membership.
    const sectionIds = [...SECTIONS].map(s => s.id)
    const menuIds    = [...MENU_ITEMS].map(m => m.id)
    expect(menuIds).toEqual(sectionIds)
  })
})

describe('Route coverage', () => {
  it.each([...MENU_ITEMS])('app/$id/page.tsx exists', ({ id }) => {
    const pagePath = resolve(__dirname, '..', 'app', id, 'page.tsx')
    expect(
      existsSync(pagePath),
      `Expected app/${id}/page.tsx to exist`,
    ).toBe(true)
  })
})

describe('RadialMenuItem href pattern', () => {
  it.each([...MENU_ITEMS])('href for $id is /$id', ({ id }) => {
    // The href is constructed as `/${item.id}` in the component.
    // Verify the id is a clean lowercase path segment.
    expect(id).toMatch(/^[a-z]+$/)
    expect(`/${id}`).toMatch(/^\/[a-z]+$/)
  })
})

describe('MENU_ITEMS unit offsets', () => {
  it('oggi (12h) has txUnit=0, tyUnit=-1', () => {
    const oggi = MENU_ITEMS.find(m => m.id === 'oggi')!
    expect(oggi.txUnit).toBeCloseTo(0, 4)
    expect(oggi.tyUnit).toBeCloseTo(-1, 4)
  })

  it('inbox (6h) has txUnit=0, tyUnit=1', () => {
    const inbox = MENU_ITEMS.find(m => m.id === 'inbox')!
    expect(inbox.txUnit).toBeCloseTo(0, 4)
    expect(inbox.tyUnit).toBeCloseTo(1, 4)
  })

  it('all items have unit-radius offset (txUnit^2 + tyUnit^2 = 1)', () => {
    for (const item of MENU_ITEMS) {
      const r = Math.sqrt(item.txUnit ** 2 + item.tyUnit ** 2)
      expect(r).toBeCloseTo(1, 4)
    }
  })
})
```

```tsx
import Link from 'next/link'

interface Props { label: string }

export function SectionPlaceholder({ label }: Props) {
  return (
    <main className="section-placeholder">
      <h1>{label}</h1>
      <p className="not-implemented">
        Questa sezione non e ancora implementata.
      </p>
      <Link href="/?view=menu" replace className="back-link">
        Torna al menu
      </Link>
    </main>
  )
}
```

The `replace` prop causes Next.js to call `router.replace` instead of `router.push`,
replacing the section URL in history rather than adding a new entry. This prevents the
Back button from cycling: after Inizia then /oggi then Torna, pressing Back goes to `/`
(Welcome), not back to `/oggi`.

```tsx
// dev/PaletteSwitcher.tsx
'use client'

/*
 * DEVELOPMENT TOOL
 * Controlled by NEXT_PUBLIC_SHOW_PALETTE_SWITCHER environment variable.
 * To remove permanently: delete this file and the conditional in app/layout.tsx.
 * The overlay can be hidden via the "Nascondi overlay" button inside the panel.
 * It is restored on page reload (no persistent storage).
 */

import { useState } from 'react'
import { PALETTES, DEFAULT_PALETTE_ID } from '@/lib/palettes'

export function PaletteSwitcher() {
  const [isVisible,      setIsVisible]      = useState(true)
  const [isExpanded,     setIsExpanded]     = useState(false)
  const [activePaletteId, setActivePaletteId] =
    useState<number>(DEFAULT_PALETTE_ID)

  // When hidden, the overlay is entirely absent from the DOM.
  // Restore by reloading the page. The active palette on <html> remains set.
  if (!isVisible) return null

  function selectPalette(id: number) {
    document.documentElement.dataset.palette = String(id)
    setActivePaletteId(id)
  }

  function handleKeyDown(e: React.KeyboardEvent) {
    if (e.key === 'Escape') {
      setIsExpanded(false)
    }
  }

  return (
    <aside
      role="region"
      aria-label="Strumenti sviluppo - selettore palette UI-08 aperta"
      className="palette-switcher"
      onKeyDown={handleKeyDown}
    >
      {isExpanded && (
        <div id="palette-switcher-panel" className="palette-switcher-panel">
          {PALETTES.map(palette => (
            <button
              key={palette.id}
              type="button"
              className="palette-switcher-item"
              aria-pressed={activePaletteId === palette.id}
              aria-label={`Palette ${palette.id}: ${palette.label} (provvisorio)`}
              onClick={() => selectPalette(palette.id)}
            >
              {palette.id}. {palette.label}
            </button>
          ))}
          {/* Hide button does not reset the active palette. Reload to restore. */}
          <button
            type="button"
            className="palette-switcher-item"
            onClick={() => setIsVisible(false)}
          >
            Nascondi overlay
          </button>
        </div>
      )}
      <button
        type="button"
        className="palette-switcher-toggle"
        aria-expanded={isExpanded}
        aria-controls="palette-switcher-panel"
        onClick={() => setIsExpanded(prev => !prev)}
      >
        P{activePaletteId}
      </button>
    </aside>
  )
}
```

The `id="palette-switcher-panel"` on the panel `<div>` matches the toggle's
`aria-controls="palette-switcher-panel"`. Both must be present.

```tsx
'use client'

import { useEffect, type RefObject } from 'react'
import Link from 'next/link'
import { APP_NAME, MENU_ITEMS } from '@/lib/constants'
import { RadialMenuItem } from './RadialMenuItem'

interface RadialMenuProps {
  visible:               boolean
  isOpen:                boolean
  onOpen:                () => void
  onClose:               () => void
  firstItemRef:          RefObject<HTMLAnchorElement | null>
  firstFallbackItemRef:  RefObject<HTMLAnchorElement | null>
  centerButtonRef:       RefObject<HTMLButtonElement | null>
}

export function RadialMenu({
  visible,
  isOpen,
  onOpen,
  onClose,
  firstItemRef,
  firstFallbackItemRef,
  centerButtonRef,
}: RadialMenuProps) {
  // Escape closes the menu only when isOpen is true.
  // If the Escape event originated inside .palette-switcher, the PaletteSwitcher
  // handles it independently (collapsing its own panel) - the radial menu must
  // not also close. Check e.target before calling onClose.
  useEffect(() => {
    function handleKeyDown(e: KeyboardEvent) {
      if (e.key === 'Escape' && isOpen) {
        const switcher = document.querySelector('.palette-switcher')
        if (switcher?.contains(e.target as Node)) return
        onClose()
      }
    }
    document.addEventListener('keydown', handleKeyDown)
    return () => document.removeEventListener('keydown', handleKeyDown)
  }, [isOpen, onClose])

  return (
    <nav
      className="screen radial-menu"
      aria-label="Navigazione principale"
      data-hidden={!visible || undefined}
      inert={!visible || undefined}
      aria-hidden={!visible || undefined}
    >
      <div className="radial-bg" aria-hidden="true" />

      {/* Items container: inert when closed so Tab reaches only the center button.
          Both .radial-items-radial and .radial-items-fallback live here and
          inherit the inert state. CSS toggles which layout is visible. */}
      <div
        className="radial-items"
        id="radial-items"
        inert={!isOpen || undefined}
      >
        {/* Radial layout: visible above the fallback breakpoint (500px provisional) */}
        <div className="radial-items-radial">
          {MENU_ITEMS.map((item, index) => (
            <RadialMenuItem
              key={item.id}
              item={item}
              index={index}
              isOpen={isOpen}
              ref={index === 0 ? firstItemRef : undefined}
            />
          ))}
        </div>

        {/* Linear fallback: visible below the fallback breakpoint.
            Same ARIA link names as the radial items.
            firstFallbackItemRef is attached to the first link so AppShell can
            focus it when the fallback layout is active.
            Breakpoint is provisional (500px) - calibrate during visual review. */}
        <ul className="radial-items-fallback" aria-label="Navigazione - lista">
          {MENU_ITEMS.map((item, index) => (
            <li key={item.id}>
              <Link
                href={`/${item.id}`}
                aria-label={item.label}
                ref={index === 0 ? firstFallbackItemRef : undefined}
              >
                {item.label}
              </Link>
            </li>
          ))}
        </ul>
      </div>

      <button
        ref={centerButtonRef}
        className="radial-center"
        type="button"
        onClick={isOpen ? onClose : onOpen}
        aria-expanded={isOpen}
        aria-controls="radial-items"
        aria-label={isOpen ? 'Chiudi menu' : 'Apri menu'}
      >
        {/* App name (visible when open) */}
        <span
          className="app-name"
          data-hidden={!isOpen || undefined}
          aria-hidden={!isOpen || undefined}
        >
          {APP_NAME}
        </span>

        {/* Hamburger (visible when closed) */}
        <span
          className="compact-icon"
          data-hidden={isOpen || undefined}
          aria-hidden={isOpen || undefined}
        >
          <svg
            width="20" height="20" viewBox="0 0 20 20"
            fill="currentColor" aria-hidden="true" focusable="false"
          >
            <rect x="2" y="4"  width="16" height="2" rx="1" />

            <rect x="2" y="14" width="16" height="2" rx="1" />
          </svg>
        </span>
      </button>
    </nav>
  )
}
```

```tsx
// components/AppShell.tsx
'use client'

import { useState, useRef, useEffect } from 'react'
import { useRouter } from 'next/navigation'
import type { AppView } from '@/lib/types'
import { WelcomeScreen } from './WelcomeScreen'
import { RadialMenu } from './RadialMenu'

interface AppShellProps {
  initialView: AppView
}

export function AppShell({ initialView }: AppShellProps) {
  const [view, setView] = useState<AppView>(initialView)
  const router          = useRouter()

  // prevViewRef distinguishes mount from genuine in-app transitions.
  // - On initial mount: prevView is null -> skip focus assignment.
  // - On Strict Mode double-invoke: prevView === view -> skip.
  // - On genuine transition: prevView !== view -> assign focus.
  const prevViewRef         = useRef<AppView | null>(null)
  const firstItemRef        = useRef<HTMLAnchorElement | null>(null)
  const firstFallbackItemRef = useRef<HTMLAnchorElement | null>(null)
  const centerButtonRef     = useRef<HTMLButtonElement | null>(null)

  // handleStart: the only transition that writes to the URL.
  // router.replace (not push) ensures Browser Back from a section lands at
  // /?view=menu rather than replaying the Welcome screen.
  function handleStart() {
    setView('MENU_OPEN')
    router.replace('/?view=menu')
  }

  // Returns the ref for the first interactive menu item based on which layout
  // is currently visible. Uses window.matchMedia to detect the active layout
  // without duplicating React state. Falls back to radial ref if window is
  // unavailable (SSR guard - this effect only runs in the browser).
  function getFirstMenuItemRef(): HTMLAnchorElement | null {
    if (typeof window === 'undefined') return firstItemRef.current
    const isFallback = window.matchMedia('(max-width: 500px)').matches
    return isFallback ? firstFallbackItemRef.current : firstItemRef.current
  }

  useEffect(() => {
    const prevView      = prevViewRef.current
    prevViewRef.current = view

    if (prevView === null) return   // initial mount - no focus
    if (prevView === view) return   // Strict Mode double-invoke - no focus

    // Genuine in-app transition: assign focus to the contextually correct element.
    if (view === 'MENU_OPEN') {
      getFirstMenuItemRef()?.focus()
    } else if (view === 'MENU_CLOSED') {
      centerButtonRef.current?.focus()
    }
  }, [view])

  return (
    <>
      <WelcomeScreen
        visible={view === 'WELCOME'}
        onStart={handleStart}
      />
      <RadialMenu
        visible={view !== 'WELCOME'}
        isOpen={view === 'MENU_OPEN'}
        onOpen={() => setView('MENU_OPEN')}
        onClose={() => setView('MENU_CLOSED')}
        firstItemRef={firstItemRef}
        firstFallbackItemRef={firstFallbackItemRef}
        centerButtonRef={centerButtonRef}
      />
    </>
  )
}
```
